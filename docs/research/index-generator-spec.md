# .ghfs/INDEX.md Generator Specification

Specification for generating the `.ghfs/INDEX.md` file during sync operations.

---

## Purpose

The `INDEX.md` file provides a human-readable and agent-friendly overview of the synced repository mirror. It should be automatically regenerated during each sync operation.

---

## Generation Timing

Generate `INDEX.md`:
- ✅ After every successful sync operation
- ✅ Include current timestamp
- ✅ Include sync statistics
- ✅ Include active configuration

---

```
.ghfs/INDEX.md
```

## Dynamic Sections

### Conditional Structure

Generate based on active config:

```typescript
const conditionalStructure = []

if (config.sync.closed) {
  conditionalStructure.push(`├── issues/closed/              (closed issues)`)
  conditionalStructure.push(`├── pulls/closed/               (closed pull requests)`)
}

if (config.sync.activity) {
  conditionalStructure.push(`├── activity.md                 (repository activity)`)
}

if (config.sync.languages) {
  conditionalStructure.push(`├── languages.json              (language breakdown)`)
}

if (config.sync.contributors) {
  conditionalStructure.push(`├── contributors.json           (contributor stats)`)
}

if (config.sync.projects) {
  conditionalStructure.push(`├── projects/                   (Projects v2)`)
  conditionalStructure.push(`│   ├── projects.json           (all projects)`)
  conditionalStructure.push(`│   └── {number}-{slug}.json    (individual projects)`)
}

if (config.sync.mergeQueue) {
  conditionalStructure.push(`├── merge-queue/                (merge queue data)`)
  conditionalStructure.push(`│   ├── status.json             (queue status)`)
  conditionalStructure.push(`│   └── entries.json            (queue entries)`)
}

if (config.sync.wiki) {
  conditionalStructure.push(`├── wiki/                       (wiki pages)`)
}

if (config.sync.discussions) {
  conditionalStructure.push(`├── discussions/                (discussion threads)`)
}

if (config.extended.sbom || config.extended.dependabotAlerts || config.extended.securityAdvisories) {
  conditionalStructure.push(`├── security/                   (security data)`)
  if (config.extended.sbom) {
    conditionalStructure.push(`│   ├── sbom.json               (SBOM)`)
  }
  if (config.extended.dependabotAlerts) {
    conditionalStructure.push(`│   ├── dependabot-alerts.json  (alerts)`)
    conditionalStructure.push(`│   ├── dependabot-summary.json (summary)`)
  }
  if (config.extended.dependencyReview) {
    conditionalStructure.push(`│   └── dependency-review/      (per-PR reviews)`)
  }
}

if (config.extended.contextPacks) {
  conditionalStructure.push(`├── context-packs/              (prompt bundles)`)
}

if (config.extended.searchIndex) {
  conditionalStructure.push(`├── search.jsonl                (search index)`)
}

if (config.extended.refsGraph) {
  conditionalStructure.push(`├── refs.json                   (cross-reference graph)`)
}

if (config.extended.meSummary) {
  conditionalStructure.push(`├── me.md                       (personal summary)`)
}

conditionalStructure.push(`└── schema/`)
conditionalStructure.push(`    └── execute.schema.json     (execute schema)`)

return conditionalStructure.join('\n')
```

### Conditional Sections

```typescript
const conditionalSections = []

if (config.sync.activity) {
  conditionalSections.push(`### Activity\n- [Repository Activity](./activity.md) - Recent push/PR/merge events`)
}

if (config.sync.languages) {
  conditionalSections.push(`### Languages\n- [Language Breakdown](./languages.json) - Repository language statistics`)

if (config.sync.contributors) {
  conditionalSections.push(`### Contributors\n- [Contributor List](./contributors.json) - Repository contributors`)
}

if (config.sync.projects) {
  conditionalSections.push(`### Projects\n- [Projects Overview](./projects/projects.json) - All Projects v2 boards`)
}

if (config.sync.mergeQueue) {
  conditionalSections.push(`### Merge Queue\n- [Queue Status](./merge-queue/status.json) - Merge queue state`)
}

if (config.sync.wiki) {
  conditionalSections.push(`### Wiki\n- [Wiki Index](./wiki/index.md) - Wiki pages`)
}

if (config.sync.discussions) {
  conditionalSections.push(`### Discussions\n- [Discussions Index](./discussions/index.md) - Discussion threads`)
}

if (config.extended.sbom || config.extended.dependabotAlerts || config.extended.securityAdvisories) {
  const securityItems = []
  if (config.extended.sbom) securityItems.push('- [SBOM](./security/sbom.json) - Software Bill of Materials')
  if (config.extended.dependabotAlerts) {
    securityItems.push('- [Dependabot Alerts](./security/dependabot-alerts.json) - Vulnerability alerts')
    securityItems.push('- [Dependabot Summary](./security/dependabot-summary.json) - Alert summary')
  }
  if (config.extended.securityAdvisories) securityItems.push('- [Security Advisories](./security/advisories.json) - Security advisories')
  if (config.extended.securitySummary) securityItems.push('- [Security Summary](./security/README.md) - Security overview')
  
  conditionalSections.push(`### Security\n${securityItems.join('\n')}`)
}

if (config.extended.meSummary || config.extended.searchIndex || config.extended.refsGraph || config.extended.contextPacks) {
  const intelligenceItems = []
  if (config.extended.meSummary) intelligenceItems.push('- [Personal Summary](./me.md) - Your issues/PRs')
  if (config.extended.searchIndex) intelligenceItems.push('- [Search Index](./search.jsonl) - Fast local search')
  if (config.extended.refsGraph) intelligenceItems.push('- [Cross-Reference Graph](./refs.json) - Issue/PR relationships')
  if (config.extended.contextPacks) intelligenceItems.push('- [Context Packs](./context-packs/) - Prompt-sized bundles')
  
  conditionalSections.push(`### Agent Intelligence\n${intelligenceItems.join('\n')}`)
}

return conditionalSections.join('\n\n')
```

## Implementation Location

Add to `src/sync/write-index.ts`:

```typescript
import { writeFile } from 'node:fs/promises'
import { join } from 'pathe'
import type { GhfsResolvedConfig } from '../types/config'
import type { SyncRepositoryResult } from './sync-repository-types'

export async function writeIndex(
  storageDirAbsolute: string,
  config: GhfsResolvedConfig,
  syncResult: SyncRepositoryResult,
): Promise<void> {
  const indexPath = join(storageDirAbsolute, 'INDEX.md')
  const content = generateIndexContent(config, syncResult)
  await writeFile(indexPath, content, 'utf8')
}
```
